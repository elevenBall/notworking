método para comportamento


Objecto escultórico, cinético e sonoro, construído a partir da reutilização de componentes recuperados de equipamentos fotográficos e de vídeo analógicos descartados.

Uma cabeça de leitura de vídeo magnética (Hi8), LEDs e um altifalante são controlados por um microcontrolador. Da dissecação, escuta e recomposição de tecnologias obsoletas, de mecanismos esquecidos.

Um pequeno corpo-máquina entre o som, o lixo e o que é existir.

código do Arduino: 
int spk = 11;
int mili = 100;
int coil1 = 8;  // Fio 1
int coil2 = 7;  // Fio 3
int coil3 = 6;  // Fio 3
int led1 = 9;
int led2 = 10;
int rnd = 0;
int ctr = 0;
int ctr2 = 0;
int prev = 0;

void setup() {
  Serial.begin(9600);
  pinMode(coil1, OUTPUT);
  pinMode(coil2, OUTPUT);
  pinMode(coil3, OUTPUT);
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
}

void loop() {
  rnd = random(36); //36
  if (rnd == prev) {
    rnd = random(36);
  };
  prev = rnd;
  Serial.println(rnd);
if (rnd == 0 && ctr2 == 0) {
    mili = random(800) + 100;
    digitalWrite(led1, LOW);
    digitalWrite(led2, HIGH);
    tone(spk, 200, mili * 0.5);
    ctr2 = 1;
    delay(mili * 0.6);
    tone(spk, 1000, mili * 0.25);
    digitalWrite(led1, HIGH);
    digitalWrite(led2, LOW);
    delay(mili * 0.25);
    tone(spk, 8000, mili * 0.25);
    digitalWrite(led1, LOW);
    digitalWrite(led2, HIGH);
    delay(mili * 0.25);
    digitalWrite(led1, LOW);
    digitalWrite(led2, LOW);
  };
  if (rnd >= 1) {
    mili = random(500) + 10;
    digitalWrite(led1, HIGH);
    tone(spk, 35, mili * 0.9);
    digitalWrite(coil1, HIGH);
    digitalWrite(coil2, LOW);
    digitalWrite(coil3, LOW);
    delay(mili);
    digitalWrite(led1, LOW);
    ctr = 0;
    ctr2 = 0;
  };
  if (rnd >= 10) {
    mili = random(500) + 10;
    tone(spk, 35, mili * 0.9);
    digitalWrite(coil1, LOW);
    digitalWrite(coil2, HIGH);
    digitalWrite(coil3, LOW);
    digitalWrite(led1, HIGH);
    delay(mili);
    digitalWrite(led1, LOW);
    ctr = 0;
    ctr2 = 0;
  };
  if (rnd >= 20) {
    mili = random(800);
    if (ctr == 0) {
      digitalWrite(led2, HIGH);
      tone(spk, 10000, 100);
    };
    digitalWrite(coil1, LOW);
    digitalWrite(coil2, LOW);
    digitalWrite(coil3, LOW);
    delay(mili * 0.5);
    digitalWrite(led2, LOW);
    delay(mili * 0.5);
    ctr = 1;
    ctr2 = 0;
  };
  if (rnd >= 30) {
    mili = random(3000) + 2000;
    delay(mili);
    ctr2 = 0;
  };
}
